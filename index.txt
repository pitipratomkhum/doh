<script>
let userEmail = "";

function unlockExam(){
  const email = document.getElementById("email").value;
  if(!email){
    alert("กรุณากรอกอีเมล");
    return;
  }
  userEmail = email;
  document.getElementById("overlay").style.display = "none";
}

const questions = [
  {q:"ถนนที่ mobility สูงสุดคืออะไร?", options:["Local","Collector","Arterial","Expressway"], answer:3},
  {q:"Superelevation ใช้ลดอะไร?", options:["แรงต้าน","แรงเหวี่ยง","ความเร็ว","slope"], answer:1},
  {q:"Reaction time มาตรฐานเท่าไร?", options:["1","1.5","2.5","3.5"], answer:2}
];

const quizDiv = document.getElementById("quiz");

questions.forEach((q,i)=>{
  const div=document.createElement("div");
  div.className="question";
  div.innerHTML = `<p>${i+1}. ${q.q}</p>` +
    q.options.map((opt,idx)=>
      `<label><input type='radio' name='q${i}' value='${idx}'> ${opt}</label><br>`
    ).join("");
  quizDiv.appendChild(div);
});

function submitQuiz(){
  let score=0;
  questions.forEach((q,i)=>{
    const ans=document.querySelector(`input[name=q${i}]:checked`);
    if(ans && parseInt(ans.value)===q.answer) score++;
  });

  document.getElementById("result").innerText =
    `คะแนนของคุณ: ${score}/${questions.length}`;

  // เก็บข้อมูล (Google Sheet)
  fetch("https://script.google.com/macros/s/AKfycbxjV9w1D1z_sZgTb4yy8uDTE0kbLFbthUVYvTYJAK-nmOgjoxxocY22LMoOkjxWpSwv/exec", {
    method:"POST",
    body: JSON.stringify({
      email:userEmail,
      score:score
    })
  });
}
</script>
